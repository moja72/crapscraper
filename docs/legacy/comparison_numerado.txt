65:def normalize_url_key(value: Any) -> str:
66:    text = _normalize_spaces(value).lower()
67:    if not text:
68:        return ""
69:
70:    if not re.match(r"^[a-z][a-z0-9+.-]*://", text):
71:        text = "https://" + text.lstrip("/")
72:
73:    try:
74:        parsed = urlparse(text)
75:    except Exception:
76:        return ""
77:
78:    host = (parsed.netloc or "").lower()
79:    if host.startswith("www."):
80:        host = host[4:]
81:
82:    path = re.sub(r"/{2,}", "/", parsed.path or "/")
83:    path = path.rstrip("/")
84:    return f"{host}{path}"

87:def clean_version(value: Any) -> str:
88:    text = _normalize_spaces(value)
89:    if not text:
90:        return ""
91:
92:    formula = re.fullmatch(r'=\"(.*)\"', text)
93:    if formula:
94:        text = formula.group(1).replace('""', '"')
95:
96:    if text.startswith("'"):
97:        text = text[1:]
98:
99:    text = re.sub(r"^\s*(vers[aã]o|version|ver\.?)[\s:#-]*", "", text, flags=re.IGNORECASE)
100:    text = text.replace(",", ".")
101:    return re.sub(r"\s+", "", text).strip(" .-_")

104:def is_suspicious_spreadsheet_version(value: str) -> bool:
105:    # Valores como 1.6.2006 e 2.2.2008 aparecem no catálogo atual e podem
106:    # ser versões convertidas por planilha. Não devem ser usados para decisão automática.

110:def _version_tokens(value: str) -> list[tuple[int, Any]]:
111:    tokens: list[tuple[int, Any]] = []
112:    for raw in re.findall(r"\d+|[a-zA-Z]+", value or ""):
113:        if raw.isdigit():
114:            tokens.append((1, int(raw)))
115:        else:
116:            tokens.append((0, raw.lower()))
117:    return tokens

120:def compare_versions(source_version: str, site_version: str) -> int | None:
121:    source = clean_version(source_version)
122:    site = clean_version(site_version)
123:    if not source or not site:
124:        return None
125:
126:    if is_suspicious_spreadsheet_version(source) or is_suspicious_spreadsheet_version(site):
127:        return None
128:
129:    left = _version_tokens(source)
130:    right = _version_tokens(site)
131:    if not left or not right:
132:        return None
133:
134:    max_len = max(len(left), len(right))
135:    left += [(1, 0)] * (max_len - len(left))
136:    right += [(1, 0)] * (max_len - len(right))
137:
138:    for l_token, r_token in zip(left, right):
139:        if l_token == r_token:
140:            continue
141:
142:        # Um número é tratado como versão estável e uma palavra como qualificador.
143:        if l_token[0] != r_token[0]:
144:            return 1 if l_token[0] > r_token[0] else -1
145:
146:        return 1 if l_token[1] > r_token[1] else -1
147:
148:    return 0

151:def _read_csv_rows(path: Path) -> list[dict[str, str]]:
152:    if not path.exists() or not path.is_file():
153:        raise FileNotFoundError(f"Arquivo não encontrado: {path}")
154:
155:    with path.open("r", encoding="utf-8-sig", newline="") as file:
156:        reader = csv.DictReader(file)
157:        rows: list[dict[str, str]] = []
158:        for raw_row in reader:
159:            row: dict[str, str] = {}
160:            for raw_key, raw_value in dict(raw_row or {}).items():
161:                key = str(raw_key or "").lstrip("\t\ufeff").strip()
162:                row[key] = "" if raw_value is None else str(raw_value)
163:            rows.append(row)
164:        return rows

270:def _build_status(source_version: str, site_version: str) -> str:
271:    if not source_version and not site_version:
272:        return "version_review"
273:    if not site_version:
274:        return "site_version_missing"
275:    if not source_version:
276:        return "source_version_missing"
277:    if source_version == site_version:
278:        return "updated"
279:    if is_suspicious_spreadsheet_version(source_version) or is_suspicious_spreadsheet_version(site_version):
280:        return "version_review"
281:
282:    comparison = compare_versions(source_version, site_version)
283:    if comparison is None:
284:        return "version_review"
285:    if comparison > 0:
286:        return "update_available"
287:    if comparison < 0:
288:        return "site_ahead"
289:    return "updated"

337:                source_index = candidate

340:        if source_index is None:
341:            result_rows.append(
342:                {
343:                    "status": "site_only",
344:                    "status_label": _STATUS_LABELS["site_only"],

347:                    "site_name": site["site_name"],
348:                    "site_version": site["site_version"],
349:                    "site_official_url": site["site_official_url"],
350:                    "site_categories": site["site_categories"],
351:                    "source_name": "",
352:                    "source_version": "",

354:                    "source_official_url": "",

397:    suspicious_site_versions = sum(

411:        "status_labels": dict(_STATUS_LABELS),

436:    *,
437:    source_path: str | Path | None = None,
438:    site_path: str | Path | None = None,
439:    status: str = "",
440:    query: str = "",
441:    page: int = 1,

444:) -> dict[str, Any]:

446:        source_path or settings.COMPARISON_ULTRAPACK_CSV_PATH

474:                        _normalize_spaces(row.get("source_version")),
475:                        _normalize_spaces(row.get("source_category")),
476:                    ]
477:                ).lower()
478:            )
479:        ]
480:
481:    resolved_page_size = max(
482:        1,
483:        min(
484:            int(page_size or settings.COMPARISON_DEFAULT_PAGE_SIZE),
485:            int(settings.COMPARISON_MAX_PAGE_SIZE),
486:        ),
487:    )
488:    total_filtered = len(rows)
489:    total_pages = max(1, (total_filtered + resolved_page_size - 1) // resolved_page_size)
490:    resolved_page = max(1, min(int(page or 1), total_pages))
491:    start = (resolved_page - 1) * resolved_page_size
492:    end = start + resolved_page_size
493:
494:    return {
495:        "ok": True,
496:        "summary": {key: value for key, value in full.items() if key != "rows"},
497:        "filters": {
498:            "status": normalized_status or "all",
499:            "query": query,
500:        },
501:        "pagination": {
502:            "page": resolved_page,
503:            "page_size": resolved_page_size,
504:            "total_rows": total_filtered,
505:            "total_pages": total_pages,
506:        },
507:        "rows": rows[start:end],
508:    }
